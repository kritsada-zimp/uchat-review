@Method
isImageWithinResizeLimit(width?: number | string, height?: number | string): boolean {
  const numericWidth = typeof width === "number" ? width : Number.parseFloat(String(width ?? ""));
  const numericHeight = typeof height === "number" ? height : Number.parseFloat(String(height ?? ""));

  if (!Number.isFinite(numericWidth) || !Number.isFinite(numericHeight) || numericWidth <= 0 || numericHeight <= 0) {
    return false;
  }

  return numericWidth <= LIMIT_IMAGE_WIDTH && numericHeight <= LIMIT_IMAGE_HEIGHT;
}

อีกจุด — Number.parseFloat ไม่ใช่ปล่อยให้ JS coerce เพราะค่ามาจาก $multipart ที่ส่งทุก field เป็น string ของเดิมรอดเพราะ "1080" <= 1920 JS แปลงให้เอง แต่พึ่ง implicit coercion กับ input ที่ควบคุมไม่ได้เป็นเรื่องเปราะ

export function toRoomFileIdString(file: unknown): string | null {
  if (typeof file === "string") return file || null;

  if (file && typeof file === "object") {
    const candidate = "_id" in file || "fileId" in file
      ? ((file as { _id?: unknown })._id ?? (file as { fileId?: unknown }).fileId)
      : file;                          // bare ObjectId

    if (candidate === null || candidate === undefined) return null;
    const normalized = String(candidate);
    return normalized && normalized !== "[object Object]" ? normalized : null;
  }

  return null;
}

const readSizeLimit = this.resolveUploadSizeLimit(fileType, canUseLargeFileUpload);
let fileBuffer: Buffer;
try {
  fileBuffer = await getStream.buffer(ctx.params, { maxBuffer: readSizeLimit });
} catch (error) {
  if (error instanceof getStream.MaxBufferError) {
    // ไม่มีใครอ่าน body ต่อ — ตัดทิ้งไม่ให้ transit สะสม chunk
    if (typeof ctx.params?.destroy === "function") ctx.params.destroy();

}

เปลี่ยนอะไรกระทบใคร
IMAGE > 20 MB ถูกปฏิเสธ เมื่อ flag ONFE ที่อัปรูปเกิน 20 MB — น่าจะไม่มี เพราะ FE บีบอัดก่อนส่ง และ branch อื่นก็ cap 20 MB อยู่แล้ว
video/file ได้ thumbnail จริง เมื่อ FE ไม่ส่ง thumbnailIdfield ที่เคยว่างจะมีค่า — shape เดิม
multipage TIFF ได้หน้าแรกเดิม animated: true ต่อทุกหน้าเป็นแถบยาว — เคสหาแทบไม่ได้ในแอปแชต

RAM — หายบางส่วน ไม่หายทั้งหมด

หายแน่นอน: request แขวน · ก็อป 4→1 · clone ที่ไม่มีใครอ่าน · libvips cache 50→32 MB · ไฟล์ที่จะโดนปฏิเสธหยุดอ่านที่เพดาน · IMAGE มีเพดาน

ยังอยู่ (ตั้งใจไม่แตะ):

  1. ยังไม่ใช่ streaming จริง — 1 request ยังถือไฟล์ทั้งก้อนใน heap แต่หลัง P8 เพดานจริงของเส้นนี้อยู่ที่ 10 MB (FILE/VIDEO) และ 20 MB (IMAGE) ไม่ใช่ระดับ 100 MB–1 GB อีก
  2. getBlurhash(fileBuffer) — ถ้า FE ไม่ส่ง blurhash ยัง sharp decode ภาพเต็มใบอีกรอบ
  3. file.putObject ฝั่ง file service ยัง clone + buffer ซ้ำ — คนละ service

เจอแต่ยังไม่แก้ — ควรตั้งใบแยก

  1. type: "FILE" bypass resize ได้โดยไม่ต้องมี feature flag — mimeTypeInsert เป็น undefined เมื่อ type === FILE ทำให้ไม่มี processor ถูกเลือก
  2. isOriginal / isLocked มาจาก $multipart ซึ่งเป็น string — uploadFileV3 เป็น @Action() เปล่า ไม่มี params validator เลย ค่า "false" เป็น truthy
  3. isOriginal=true + FE ไม่ส่ง width/height → undefined ถาวรใน DB และ HEIC ขึ้น S3 ดิบ ๆ
  4. stale spec 23 ตัว

บทเรียนที่เอาไปใช้ต่อได้

  1. stream มี consumer ได้คนเดียว — ต้องใช้หลายที่ ให้ buffer ก้อนเดียว แล้ว Readable.from() ใหม่ทุกครั้ง
  2. sharp instance ใช้ได้ท่าเดียว — เผลอทำทั้งสอง อาการไม่ใช่ error แต่เป็นแขวนเงียบ ๆ ซึ่งหายากกว่ามาก
  3. log ต้องอยู่หลังจุดตัดสินใจ — log ก่อนเช็คเงื่อนไขทำให้ทีมไล่ผิดทาง
  4. อย่าตัดสินใจจากค่าที่ client แจ้ง — cap สิ่งที่เราทำ ไม่ใช่เชื่อตัวเลขจาก client
  5. วัด regression ด้วยการ diff รายชื่อเทส ไม่ใช่นับจำนวน
